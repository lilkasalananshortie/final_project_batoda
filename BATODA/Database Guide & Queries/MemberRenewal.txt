CREATE TABLE MemberRenewal (
    RenewalID INT IDENTITY(1,1) PRIMARY KEY,
    BodyNumber INT NOT NULL,
    DateRenewed DATE NOT NULL,  -- copy MemberInfo == conflict kasi not PK
    ExpiryDate AS DATEADD(YEAR, 1, DateRenewed) PERSISTED,
    FOREIGN KEY (BodyNumber) REFERENCES MemberInfo(BodyNumber)
);
